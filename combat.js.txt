class CombatEngine{
 constructor(heroes,enemies,onChange){this.heroes=heroes;this.enemies=enemies;this.onChange=onChange;this.state="INITIATIVE";this.round=1;this.turnIndex=0;this.initiative=[];this.log=[];this.pending=null;this.responseAction=null}
 rollD10(){return Math.floor(Math.random()*10)+1} all(){return[...this.heroes,...this.enemies]} living(s){return(s==="hero"?this.heroes:this.enemies).filter(u=>u.alive)} getUnit(id){return this.all().find(u=>u.id===id)} current(){const e=this.initiative[this.turnIndex];return e?this.getUnit(e.id):null} notify(){this.onChange?.()} addLog(x){this.log.unshift(x);this.log=this.log.slice(0,60)}
 start(){this.heroes.forEach(h=>{h.hp=h.maxHp;h.alive=true;h.tapped=false;h.mana=0});this.initiative=this.all().map((u,i)=>({id:u.id,side:u.side,roll:this.rollD10()+(u.keywords?.includes("Haste")?5:0),i})).sort((a,b)=>b.roll-a.roll||a.i-b.i);this.addLog("Initiative rolled.");this.beginTurn()}
 beginTurn(){if(this.checkEnd())return;if(this.turnIndex>=this.initiative.length){this.turnIndex=0;this.round++;this.addLog(`Round ${this.round} begins.`)}const u=this.current();if(!u||!u.alive){this.advance();return}if(u.side==="hero"){u.tapped=false;u.mana++;this.state="PLAYER_TURN";this.addLog(`${u.name} untaps and gains 1 mana.`);this.notify()}else{this.state="ENEMY_TURN";this.addLog(`${u.name}'s turn.`);this.notify();setTimeout(()=>this.prepareEnemyAction(u),450)}}
 canAct(h){return h.alive&&!h.tapped}
 chooseAttack(){if(this.state!=="PLAYER_TURN"||!this.canAct(this.current()))return;this.pending={type:"attack"};this.state="TARGET_ENEMY";this.notify()}
 chooseSpell(spellId){const h=this.current(),s=h.spells.find(x=>x.id===spellId);if(this.state!=="PLAYER_TURN"||!this.canAct(h)||!s||h.mana<s.cost)return;this.pending={type:"spell",spell:s};if(s.effect==="damage")this.state="TARGET_ENEMY";else if(s.effect==="reanimate")this.state="TARGET_GRAVE";this.notify()}
 chooseAbility(abilityId){const h=this.current(),a=h.abilities?.find(x=>x.id===abilityId);if(this.state!=="PLAYER_TURN"||!this.canAct(h)||!a)return;this.pending={type:"ability",ability:a};this.state="TARGET_HERO";this.notify()}
 cancel(){this.pending=null;this.state="PLAYER_TURN";this.notify()}
 selectTarget(id){const actor=this.current(),target=this.getUnit(id);if(!this.pending||!actor||!target)return;if(this.pending.type==="attack"){this.damage(actor,target,actor.power);if(!actor.keywords?.includes("Vigilance"))actor.tapped=true}else if(this.pending.type==="spell"){const s=this.pending.spell;actor.mana-=s.cost;actor.tapped=true;if(s.effect==="damage")this.damage(actor,target,s.amount);if(s.effect==="reanimate"){target.alive=true;target.hp=target.maxHp;target.tapped=true;this.addLog(`${actor.name} reanimates ${target.name}. It returns tapped.`)}}else if(this.pending.type==="ability"){actor.tapped=true;target.mana+=this.pending.ability.amount;this.addLog(`${actor.name} adds ${this.pending.ability.amount} mana to ${target.name}.`)}this.pending=null;this.notify();if(!this.checkEnd())this.advance()}
 damage(a,t,n){t.hp=Math.max(0,t.hp-n);this.addLog(`${a.name} deals ${n} damage to ${t.name}.`);if(t.hp<=0){t.alive=false;this.addLog(`${t.name} is defeated.`);if(a.side==="hero"&&t.side==="enemy"){a.xp+=t.manaValue;this.addLog(`${a.name} gains ${t.manaValue} XP.`)}}}
 pass(){if(this.state!=="PLAYER_TURN")return;this.addLog(`${this.current().name} passes and remains ${this.current().tapped?"tapped":"untapped"}.`);this.advance()}
 prepareEnemyAction(enemy){const targets=this.living("hero");if(!targets.length)return this.checkEnd();const weighted=[];targets.forEach(h=>{const w=h.tapped?1:3;for(let i=0;i<w;i++)weighted.push(h)});const target=weighted[Math.floor(Math.random()*weighted.length)];const isSpell=enemy.spell&&Math.random()<.45;this.responseAction={enemy,target,type:isSpell?"spell":"attack",name:isSpell?enemy.spell.name:"Attack",damage:isSpell?enemy.spell.amount:enemy.power};const responses=this.availableResponses(this.responseAction);if(responses.length){this.state="RESPONSE";this.notify()}else this.resolveEnemyAction()}
 availableResponses(action){const out=[];this.heroes.forEach(h=>{if(!h.alive||h.tapped)return;h.spells.forEach(s=>{if(h.mana<s.cost||s.speed!=="Instant")return;if(s.effect==="counter"&&action.type==="spell")out.push({hero:h,spell:s});if(s.effect==="damage")out.push({hero:h,spell:s})})});return out}
 useResponse(heroId,spellId){if(this.state!=="RESPONSE")return;const h=this.getUnit(heroId),s=h.spells.find(x=>x.id===spellId);if(!h||!s||h.tapped||h.mana<s.cost)return;h.mana-=s.cost;h.tapped=true;if(s.effect==="counter"){this.addLog(`${h.name} casts ${s.name} and counters ${this.responseAction.name}.`);this.responseAction=null;this.advance();return}if(s.effect==="damage"){this.addLog(`${h.name} responds with ${s.name}.`);this.damage(h,this.responseAction.enemy,s.amount);if(!this.responseAction.enemy.alive){this.responseAction=null;if(!this.checkEnd())this.advance();return}}this.resolveEnemyAction()}
 declineResponse(){if(this.state==="RESPONSE")this.resolveEnemyAction()}
 resolveEnemyAction(){const a=this.responseAction;if(!a)return;if(a.enemy.alive&&a.target.alive){a.target.hp=Math.max(0,a.target.hp-a.damage);this.addLog(`${a.enemy.name} uses ${a.name} on ${a.target.name} for ${a.damage} damage.`);if(a.target.hp<=0){a.target.alive=false;this.addLog(`${a.target.name} falls for this encounter.`)}}this.responseAction=null;this.notify();if(!this.checkEnd())setTimeout(()=>this.advance(),300)}
 advance(){if(this.checkEnd())return;this.turnIndex++;this.state="TURN_START";this.beginTurn()}
 checkEnd(){if(!this.living("enemy").length){this.state="VICTORY";this.heroes.forEach(h=>{h.alive=true;h.hp=h.maxHp;h.tapped=false});this.addLog("Victory. Heroes return at full health.");this.notify();return true}if(!this.living("hero").length){this.state="DEFEAT";this.addLog("Party defeated.");this.notify();return true}return false}
}
