let engine;const app=document.getElementById("app");const esc=v=>String(v).replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#039;"}[c]));
function start(){engine=new CombatEngine(DekQuestData.heroes(),DekQuestData.enemies(),render);engine.start()}
function unit(u,current){const spells=(u.spells||[]).map(s=>`${s.name} ${s.cost}`).join(", ");return `<div class="unit ${u.alive?'':'dead'} ${current?'current':''} ${u.tapped?'tapped':''}"><div class="row"><strong class="${u.side==='hero'?'hero-name':'enemy-name'}">${esc(u.name)}</strong><span>${u.alive?`${u.hp}/${u.maxHp} HP`:'Defeated'}</span></div><div class="meta">Power ${u.power}${u.side==='hero'?` | Mana ${u.mana} | ${u.tapped?'Tapped':'Untapped'}`:` | MV ${u.manaValue}`}</div>${spells?`<div class="meta spell">${esc(spells)}</div>`:''}</div>`}
function actionPanel(){const u=engine.current();if(engine.state==="PLAYER_TURN"){const spellButtons=u.spells.map(s=>`<button ${u.mana<s.cost?'disabled':''} onclick="engine.chooseSpell('${s.id}')">${esc(s.name)} (${s.cost})</button>`).join("");const abilityButtons=(u.abilities||[]).map(a=>`<button onclick="engine.chooseAbility('${a.id}')">${esc(a.name)}</button>`).join("");return `<h2>${esc(u.name)}'s Turn</h2><div class="actions"><button onclick="engine.chooseAttack()">Attack (${u.power})</button>${spellButtons}${abilityButtons}<button onclick="engine.pass()">Pass</button></div>`}if(engine.state==="TARGET_ENEMY")return targetList(engine.living("enemy"),"Choose an enemy target");if(engine.state==="TARGET_HERO")return targetList(engine.living("hero"),"Choose whose mana pool receives the mana");if(engine.state==="TARGET_GRAVE")return targetList(engine.all().filter(x=>!x.alive),"Choose a dead hero or enemy to reanimate");if(engine.state==="RESPONSE"){const a=engine.responseAction,res=engine.availableResponses(a);return `<div class="response panel"><h2>Instant Response</h2><p>${esc(a.enemy.name)} will use ${esc(a.name)} on ${esc(a.target.name)}.</p><div class="choice-list">${res.map(r=>`<button onclick="engine.useResponse('${r.hero.id}','${r.spell.id}')">${esc(r.hero.name)}: ${esc(r.spell.name)} (${r.spell.cost})</button>`).join("")}</div><button class="secondary" onclick="engine.declineResponse()">No Response</button></div>`}if(engine.state==="ENEMY_TURN")return `<h2>${esc(u.name)}'s Turn</h2><p class="warning">Enemy choosing an action...</p>`;if(engine.state==="VICTORY")return `<h2 class="good">Victory</h2><button onclick="start()">New Test Encounter</button>`;if(engine.state==="DEFEAT")return `<h2 class="warning">Defeat</h2><button onclick="start()">Retry</button>`;return `<h2>Preparing...</h2>`}
function targetList(items,title){return `<h2>${title}</h2><div class="target-list">${items.map(x=>`<button class="target-button" onclick="engine.selectTarget('${x.id}')"><span>${esc(x.name)}</span><span>${x.hp}/${x.maxHp}</span></button>`).join("")}</div><button class="secondary" onclick="engine.cancel()">Cancel</button>`}
function render(){const cur=engine.current(),active=cur?.id||"";const init=engine.initiative.map((e,i)=>{const u=engine.getUnit(e.id),on=i===engine.turnIndex&&!['VICTORY','DEFEAT'].includes(engine.state);return `<div class="initiative-item ${e.side} ${on?'active':''}">${on?'▶ ':''}${esc(u.name)} (${e.roll})${u.alive?'':' ✕'}</div>`}).join("");app.innerHTML=`<section class="panel"><h2>Round ${engine.round}: Initiative</h2><div class="initiative">${init}</div></section><div class="grid"><section class="panel"><h2>Party</h2>${engine.heroes.map(h=>unit(h,h.id===active)).join("")}</section><section class="panel"><h2>Enemies</h2>${engine.enemies.map(e=>unit(e,e.id===active)).join("")}</section><section class="panel full">${actionPanel()}</section><section class="panel full"><h2>Combat Log</h2><div class="log">${engine.log.map(l=>`<div>${esc(l)}</div>`).join("")}</div></section></div>`}
document.getElementById("newRunButton").addEventListener("click",start);start();
